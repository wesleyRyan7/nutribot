const express = require('express'); const cors = require('cors'); const { GoogleGenerativeAI } = require("@google/generative-ai"); require('dotenv').config(); const app = express(); app.use(cors()); app.use(express.json({ limit: '50mb' })); const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY); app.post('/analisar', async (req, res) => { try { const { imageBase64 } = req.body; if (!imageBase64) return res.status(400).json({ erro: "Faltou a imagem!" }); const cleanBase64 = imageBase64.includes(',') ? imageBase64.split(',')[1] : imageBase64; // FORÇANDO A VERSÃO V1 PARA EVITAR O 404 const model = genAI.getGenerativeModel({ model: "gemini-1.5-flash" }, { apiVersion: 'v1' }); const result = await model.generateContent([ "Descreva os alimentos e calorias desta imagem.", { inlineData: { data: cleanBase64, mimeType: "image/jpeg" } } ]); const response = await result.response; res.json({ resposta: response.text() }); } catch (error) { console.error("ERRO:", error.message); res.status(500).json({ erro: error.message }); } }); app.listen(3000, () => console.log("NutriBot online na 3000"));
